# pythagoras calculates the length of hypotenuse c given a and b
def pythagoras(a, b):
    return sqrt(a * a + b * b)

# convert to three-address code
def pythagoras(a, b):
    asq = a * a
    bsq = b * b
    csq = asq + bsq
    c = sqrt(csq)
    return c
